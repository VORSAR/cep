**S2 — `gate/runner/sprintid.py` (lane D → lane A's `marker`/`notbuilt`/`sprints` parsers; S1.1c-2's `todocheck`).** Pure: strings in, values out, no I/O.
```python
@dataclass(frozen=True)
class ID:          # S<stage>[.<part>][<letter>][-<n>]  (sprint-plan §3)
    stage: int     # S3 → 3
    part: int      # S3.2 → 2; 0 when absent
    letter: str    # S3.2a → "a"; "" when absent
    n: int         # S3.3-2 → 2; 0 when absent (a plan check's split)

def parse(text: str) -> ID            # "S3.3-2" → ID(3, 3, "", 2); "S10" → ID(10, 0, "", 0); "s3.3", "S3.3-", "S3.3-2x", "S03.3", "S3.3-0", " S1.1c-1", "" → ValueError naming the text
def canonical(id: ID) -> str          # round-trips parse
def row_of(id: ID) -> str             # the card id: ID for "S3.3-2" → "S3.3"; "S1.1a-1" → "S1.1a"

@dataclass(frozen=True)
class Row:   id: str; closed: bool    # id is a printed row id ("S3.3", "S1.1a-1"); closed when the plan's row carries ✓
class Order:                          # the rows of gate/sprints.json in printed order
    rows: tuple[Row, ...]
    def index(self, text: str) -> int           # the printed id's position if present, else row_of(parse(text))'s; ValueError when neither is a row
    def reached(self, entry: str, current: str) -> bool   # index(entry) <= index(current): the entry's sprint has arrived
    def first_open(self) -> str                 # the first row not closed; ValueError when every row is closed
def load_order(text: str) -> Order    # gate/sprints.json: {"schema": 1, "rows": [{"id": "S1.1a-1", "closed": true}, …]}; a bad id, a duplicate, or no rows → ValueError
def plan_rows(markdown: str) -> list[Row]   # sprint-plan.md's §3 table — used ONLY by the T2 pin test: the header row pinned verbatim, a struck row skipped, the three ✓ forms recognised, any other mark or an unparseable id → ValueError
def read_marker(text: str) -> ID      # gate/current-sprint: one line, exactly one trailing LF; anything else → ValueError
```
